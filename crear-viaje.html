async function crearViaje() { const btn = document.querySelector('button[onclick="crearViaje()"]'); const textoOriginal = btn.innerText; btn.innerText = "GUARDANDO..."; btn.disabled = true; const paradas = Array.from(document.querySelectorAll('.parada-extra')).map(i => i.value).filter(v => v !== ""); // Mapeo ajustado EXACTAMENTE a tus capturas de Supabase const datosViaje = { colaboracion: document.getElementById('colaboracion').value, chofer_id: choferData ? choferData.id : null, cliente_telefono: document.getElementById('tel-cliente').value, // Ajustado precio: parseFloat(document.getElementById('precio-final').value) || 0, pasajeros: parseInt(document.getElementById('pax').value) || 1, // Ajustado de 'pax' a 'pasajeros' origen: document.getElementById('origen').value, paradas_intermedias: paradas.join(", "), // Ajustado destino: document.getElementById('destino').value, grupo_whatsapp: document.getElementById('grupo').value, // Ajustado pago_transferencia: document.getElementById('pago_transf').checked, precio_cliente: document.getElementById('precio_cliente').checked, aire_acondicionado: document.getElementById('aire').checked, ida_vuelta: document.getElementById('ida_vuelta').checked, es_reservado: document.getElementById('reservado').checked, fecha_reserva: document.getElementById('fecha-reserva').value || null, observaciones: document.getElementById('observacion').value, // Agregué estos dos que vi en tus capturas para que no den error distancia_km: parseFloat(document.getElementById('distancia').value) || 0, tarifa_km: parseFloat(document.getElementById('tarifa').value) || 0 }; try { const { data, error } = await supabaseClient .from('viajes') .insert([datosViaje]); if (error) throw error; alert("✅ ¡Viaje creado con éxito!"); window.location.href = "panel.html"; } catch (error) { console.error("Error:", error); alert("Error: " + error.message); btn.innerText = textoOriginal; btn.disabled = false; } }
